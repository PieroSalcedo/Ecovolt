DROP TABLE IF EXISTS public.energy_reading CASCADE;
DROP TABLE IF EXISTS public.device CASCADE;
DROP TABLE IF EXISTS public.energy_goal CASCADE;
DROP TABLE IF EXISTS public.room CASCADE;
DROP TABLE IF EXISTS public.home CASCADE;
DROP TABLE IF EXISTS public.users_has_roles CASCADE;
DROP TABLE IF EXISTS public.users CASCADE;
DROP TABLE IF EXISTS public.rol_has_option CASCADE;
DROP TABLE IF EXISTS public.subscription_plan CASCADE;
DROP TABLE IF EXISTS public.data_catalog CASCADE;
DROP TABLE IF EXISTS public."role" CASCADE;
DROP TABLE IF EXISTS public."option" CASCADE;
DROP TABLE IF EXISTS public."catalog" CASCADE;

-- =========================================================
-- 2. CREACION DE TABLAS
-- =========================================================
CREATE TABLE public."catalog" (
    id_catalog int4 GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    description varchar(100) NOT NULL
);

CREATE TABLE public."option" (
    id_option int4 GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    status int4 DEFAULT 1,
    "type" int4 NULL,
    "name" varchar(100) NULL,
    route varchar(255) NULL
);

CREATE TABLE public."role" (
    id_role int4 GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    status int4 DEFAULT 1,
    "name" varchar(50) NOT NULL UNIQUE
);

CREATE TABLE public.data_catalog (
    id_data_catalog int4 GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    id_catalog int4 NOT NULL REFERENCES public."catalog"(id_catalog),
    description varchar(200) NOT NULL,
    status int4 NOT NULL DEFAULT 1,
    created_at timestamp(6) DEFAULT CURRENT_TIMESTAMP,
    updated_at timestamp(6) NULL,
    created_by varchar(50) NULL,
    updated_by varchar(50) NULL
);

CREATE TABLE public.rol_has_option (
    id_option int4 NOT NULL REFERENCES public."option"(id_option),
    id_role int4 NOT NULL REFERENCES public."role"(id_role),
    PRIMARY KEY (id_option, id_role)
);

CREATE TABLE public.subscription_plan (
    id_plan int4 GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    "name" varchar(50) NOT NULL,
    device_limit int4 NOT NULL,
    monthly_price numeric(10, 2) NULL,
    billing_cycle varchar(20) NULL,
    id_support_level int4 NULL REFERENCES public.data_catalog(id_data_catalog),
    status int4 NOT NULL DEFAULT 1,
    created_at timestamp(6) DEFAULT CURRENT_TIMESTAMP,
    updated_at timestamp(6) NULL,
    created_by varchar(50) NULL,
    updated_by varchar(50) NULL
);

CREATE TABLE public.users (
    id_user int8 GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    login varchar(50) NOT NULL UNIQUE,
    password varchar(255) NOT NULL,
    email varchar(100) NOT NULL UNIQUE,
    first_name varchar(100) NOT NULL,
    last_name varchar(100) NOT NULL,
    phone_number varchar(20) NULL,
    id_plan int4 NOT NULL REFERENCES public.subscription_plan(id_plan),
    status int4 NOT NULL DEFAULT 1,
    created_at timestamp(6) DEFAULT CURRENT_TIMESTAMP,
    updated_at timestamp(6) NULL,
    created_by varchar(50) NULL,
    updated_by varchar(50) NULL
);

CREATE TABLE public.users_has_roles (
    id_rol int4 NOT NULL REFERENCES public."role"(id_role),
    id_user int8 NOT NULL REFERENCES public.users(id_user),
    PRIMARY KEY (id_rol, id_user)
);

CREATE TABLE public.home (
    id_home int8 GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    alias varchar(50) NULL,
    address text NULL,
    city varchar(100) NULL,
    square_meters int4 NULL,
    energy_tariff numeric(10, 4) NULL,
    id_property_type int4 NULL REFERENCES public.data_catalog(id_data_catalog),
    id_user int8 NOT NULL REFERENCES public.users(id_user),
    status int4 NOT NULL DEFAULT 1,
    created_at timestamp(6) DEFAULT CURRENT_TIMESTAMP,
    updated_at timestamp(6) NULL,
    created_by varchar(50) NULL,
    updated_by varchar(50) NULL
);

CREATE TABLE public.room (
    id_room int8 GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    "name" varchar(100) NOT NULL,
    floor_number int4 NULL,
    area_sqm numeric(10, 2) NULL,
    orientation varchar(20) NULL,
    id_room_type int4 NULL REFERENCES public.data_catalog(id_data_catalog),
    id_home int8 NOT NULL REFERENCES public.home(id_home),
    status int4 NOT NULL DEFAULT 1,
    created_at timestamp(6) DEFAULT CURRENT_TIMESTAMP,
    updated_at timestamp(6) NULL,
    created_by varchar(50) NULL,
    updated_by varchar(50) NULL
);

CREATE TABLE public.device (
    id_device int8 GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    "name" varchar(100) NULL,
    serial_number varchar(100) NOT NULL UNIQUE,
    manufacturer varchar(100) NULL,
    firmware_version varchar(50) NULL,
    id_category int4 NULL REFERENCES public.data_catalog(id_data_catalog),
    id_room int8 NOT NULL REFERENCES public.room(id_room),
    status int4 NOT NULL DEFAULT 1,
    created_at timestamp(6) DEFAULT CURRENT_TIMESTAMP,
    updated_at timestamp(6) NULL,
    created_by varchar(50) NULL,
    updated_by varchar(50) NULL
);

CREATE TABLE public.energy_goal (
    id_goal int4 GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    monthly_limit_kwh numeric(10, 2) NULL,
    alert_threshold_percentage int4 DEFAULT 80,
    id_home int8 NULL REFERENCES public.home(id_home),
    id_room int8 NULL REFERENCES public.room(id_room),
    id_device int8 NULL REFERENCES public.device(id_device),
    status int4 NOT NULL DEFAULT 1,
    created_at timestamp(6) DEFAULT CURRENT_TIMESTAMP,
    updated_at timestamp(6) NULL,
    created_by varchar(50) NULL,
    updated_by varchar(50) NULL
);

CREATE TABLE public.energy_reading (
    id_reading int8 GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    wattage numeric(12, 4) NOT NULL,
    voltage numeric(10, 2) NULL,
    id_device int8 NOT NULL REFERENCES public.device(id_device),
    reading_at timestamp(6) DEFAULT CURRENT_TIMESTAMP,
    status int4 DEFAULT 1
);
