-- 1. CREACIÓN DEL ESQUEMA
CREATE SCHEMA IF NOT EXISTS public;

-- 2. TABLAS BASE INDEPENDIENTES
CREATE TABLE public."catalog" (
	id_catalog int4 GENERATED BY DEFAULT AS IDENTITY(INCREMENT BY 1 MINVALUE 1 MAXVALUE 2147483647 START 1 CACHE 1 NO CYCLE) NOT NULL,
	description varchar(100) NOT NULL,
	CONSTRAINT catalog_pkey PRIMARY KEY (id_catalog)
);

CREATE TABLE public."option" (
	id_option int4 GENERATED BY DEFAULT AS IDENTITY(INCREMENT BY 1 MINVALUE 1 MAXVALUE 2147483647 START 1 CACHE 1 NO CYCLE) NOT NULL,
	status int4 NULL,
	"type" int4 NULL,
	"name" varchar(100) NULL,
	route varchar(255) NULL,
	CONSTRAINT option_pkey PRIMARY KEY (id_option)
);

CREATE TABLE public."role" (
	id_role int4 GENERATED BY DEFAULT AS IDENTITY(INCREMENT BY 1 MINVALUE 1 MAXVALUE 2147483647 START 1 CACHE 1 NO CYCLE) NOT NULL,
	status int4 NULL,
	"name" varchar(50) NOT NULL,
	CONSTRAINT role_name_key UNIQUE (name),
	CONSTRAINT role_pkey PRIMARY KEY (id_role)
);

-- 3. TABLAS DEPENDIENTES DE CATALOG / ROLE / OPTION
CREATE TABLE public.data_catalog (
	id_catalog int4 NOT NULL,
	id_data_catalog int4 GENERATED BY DEFAULT AS IDENTITY(INCREMENT BY 1 MINVALUE 1 MAXVALUE 2147483647 START 1 CACHE 1 NO CYCLE) NOT NULL,
	status int4 NOT NULL,
	created_at timestamp(6) NULL,
	updated_at timestamp(6) NULL,
	created_by varchar(50) NULL,
	updated_by varchar(50) NULL,
	description varchar(200) NOT NULL,
	CONSTRAINT data_catalog_pkey PRIMARY KEY (id_data_catalog),
	CONSTRAINT fk4iqg3ryk6rjq59jg3k5vhw831 FOREIGN KEY (id_catalog) REFERENCES public."catalog"(id_catalog)
);

CREATE TABLE public.rol_has_option (
	id_option int4 NOT NULL,
	id_role int4 NOT NULL,
	CONSTRAINT rol_has_option_pkey PRIMARY KEY (id_option, id_role),
	CONSTRAINT fkh8an0fgfde8442y7r8a1acijy FOREIGN KEY (id_role) REFERENCES public."role"(id_role),
	CONSTRAINT fkqcob1udkyqjdgiyo6tl88hu79 FOREIGN KEY (id_option) REFERENCES public."option"(id_option)
);

-- 4. PLANES DE SUSCRIPCIÓN Y USUARIOS
CREATE TABLE public.subscription_plan (
	device_limit int4 NOT NULL,
	id_plan int4 GENERATED BY DEFAULT AS IDENTITY(INCREMENT BY 1 MINVALUE 1 MAXVALUE 2147483647 START 1 CACHE 1 NO CYCLE) NOT NULL,
	id_support_level int4 NULL,
	monthly_price numeric(10, 2) NULL,
	status int4 NOT NULL,
	created_at timestamp(6) NULL,
	updated_at timestamp(6) NULL,
	billing_cycle varchar(20) NULL,
	created_by varchar(50) NULL,
	"name" varchar(50) NOT NULL,
	updated_by varchar(50) NULL,
	CONSTRAINT subscription_plan_pkey PRIMARY KEY (id_plan),
	CONSTRAINT fk3il1gvwrk95oax8oqrag2g3rn FOREIGN KEY (id_support_level) REFERENCES public.data_catalog(id_data_catalog)
);

CREATE TABLE public.users (
	id_plan int4 NOT NULL,
	status int4 NOT NULL,
	created_at timestamp(6) NULL,
	id_user int8 GENERATED BY DEFAULT AS IDENTITY(INCREMENT BY 1 MINVALUE 1 MAXVALUE 9223372036854775807 START 1 CACHE 1 NO CYCLE) NOT NULL,
	updated_at timestamp(6) NULL,
	phone_number varchar(20) NULL,
	created_by varchar(50) NULL,
	login varchar(50) NOT NULL,
	updated_by varchar(50) NULL,
	email varchar(100) NOT NULL,
	first_name varchar(100) NOT NULL,
	last_name varchar(100) NOT NULL,
	"password" varchar(255) NOT NULL,
	CONSTRAINT users_email_key UNIQUE (email),
	CONSTRAINT users_login_key UNIQUE (login),
	CONSTRAINT users_pkey PRIMARY KEY (id_user),
	CONSTRAINT fko4wxj2n7y0680cv7bip0rqyly FOREIGN KEY (id_plan) REFERENCES public.subscription_plan(id_plan)
);

CREATE TABLE public.users_has_roles (
	id_rol int4 NOT NULL,
	id_user int8 NOT NULL,
	CONSTRAINT users_has_roles_pkey PRIMARY KEY (id_rol, id_user),
	CONSTRAINT fk5hxna3dgl42i4bqchlxjfia1 FOREIGN KEY (id_rol) REFERENCES public."role"(id_role),
	CONSTRAINT fkkbt15ct6jomw09myornjytr6f FOREIGN KEY (id_user) REFERENCES public.users(id_user)
);

-- 5. ENTIDADES DE LA VIVIENDA (HOME & ROOM)
CREATE TABLE public.home (
	energy_tariff numeric(10, 4) NULL,
	id_property_type int4 NULL,
	square_meters int4 NULL,
	status int4 NOT NULL,
	created_at timestamp(6) NULL,
	id_home int8 GENERATED BY DEFAULT AS IDENTITY(INCREMENT BY 1 MINVALUE 1 MAXVALUE 9223372036854775807 START 1 CACHE 1 NO CYCLE) NOT NULL,
	id_user int8 NOT NULL,
	updated_at timestamp(6) NULL,
	alias varchar(50) NULL,
	created_by varchar(50) NULL,
	updated_by varchar(50) NULL,
	city varchar(100) NULL,
	address text NULL,
	CONSTRAINT home_pkey PRIMARY KEY (id_home),
	CONSTRAINT fkk6ibjmgne9iu7wqrsjr5491ei FOREIGN KEY (id_property_type) REFERENCES public.data_catalog(id_data_catalog),
	CONSTRAINT fktb09n02o3ajy5bfcsbyq0ey1k FOREIGN KEY (id_user) REFERENCES public.users(id_user)
);

CREATE TABLE public.room (
	area_sqm numeric(10, 2) NULL,
	floor_number int4 NULL,
	id_room_type int4 NULL,
	status int4 NOT NULL,
	created_at timestamp(6) NULL,
	id_home int8 NOT NULL,
	id_room int8 GENERATED BY DEFAULT AS IDENTITY(INCREMENT BY 1 MINVALUE 1 MAXVALUE 9223372036854775807 START 1 CACHE 1 NO CYCLE) NOT NULL,
	updated_at timestamp(6) NULL,
	orientation varchar(20) NULL,
	created_by varchar(50) NULL,
	updated_by varchar(50) NULL,
	"name" varchar(100) NOT NULL,
	CONSTRAINT room_pkey PRIMARY KEY (id_room),
	CONSTRAINT fk9jv6w3l1oefl229weveogd3n9 FOREIGN KEY (id_home) REFERENCES public.home(id_home),
	CONSTRAINT fknbppdy3eg389u2p2tlhl5yuig FOREIGN KEY (id_room_type) REFERENCES public.data_catalog(id_data_catalog)
);

-- 6. DISPOSITIVOS, METAS Y LECTURAS (DEVICES, GOALS & READINGS)
CREATE TABLE public.energy_goal (
	alert_threshold_percentage int4 NULL,
	id_goal int4 GENERATED BY DEFAULT AS IDENTITY(INCREMENT BY 1 MINVALUE 1 MAXVALUE 2147483647 START 1 CACHE 1 NO CYCLE) NOT NULL,
	monthly_limit_kwh numeric(10, 2) NULL,
	status int4 NOT NULL,
	created_at timestamp(6) NULL,
	id_home int8 NOT NULL,
	updated_at timestamp(6) NULL,
	created_by varchar(50) NULL,
	updated_by varchar(50) NULL,
	CONSTRAINT energy_goal_pkey PRIMARY KEY (id_goal),
	CONSTRAINT fk9ywcc61k4cyd18chyow7mjjpl FOREIGN KEY (id_home) REFERENCES public.home(id_home)
);

CREATE TABLE public.device (
	id_category int4 NULL,
	status int4 NOT NULL,
	created_at timestamp(6) NULL,
	id_device int8 GENERATED BY DEFAULT AS IDENTITY(INCREMENT BY 1 MINVALUE 1 MAXVALUE 9223372036854775807 START 1 CACHE 1 NO CYCLE) NOT NULL,
	id_room int8 NOT NULL,
	updated_at timestamp(6) NULL,
	created_by varchar(50) NULL,
	firmware_version varchar(50) NULL,
	updated_by varchar(50) NULL,
	manufacturer varchar(100) NULL,
	"name" varchar(100) NULL,
	serial_number varchar(100) NOT NULL,
	CONSTRAINT device_pkey PRIMARY KEY (id_device),
	CONSTRAINT device_serial_number_key UNIQUE (serial_number),
	CONSTRAINT fkbxuexq394fvawd223uajqalqf FOREIGN KEY (id_room) REFERENCES public.room(id_room),
	CONSTRAINT fkl1has14f86n6sw0x31uqi8fe1 FOREIGN KEY (id_category) REFERENCES public.data_catalog(id_data_catalog)
);

CREATE TABLE public.energy_reading (
	status int4 NULL,
	voltage numeric(10, 2) NULL,
	wattage numeric(12, 4) NOT NULL,
	id_device int8 NOT NULL,
	id_reading int8 GENERATED BY DEFAULT AS IDENTITY(INCREMENT BY 1 MINVALUE 1 MAXVALUE 9223372036854775807 START 1 CACHE 1 NO CYCLE) NOT NULL,
	reading_at timestamp(6) NULL,
	CONSTRAINT energy_reading_pkey PRIMARY KEY (id_reading),
	CONSTRAINT fkctlidwuqld85gcgdlle6dxktf FOREIGN KEY (id_device) REFERENCES public.device(id_device)
);
